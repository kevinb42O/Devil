 /** Resolve only after all damage: a team wipe always beats a pending revival. */
 private respawnCrew(){
  const survivors=this.crew.filter(p=>p.alive&&p.cause!=='DISCONNECTED');
  for(const p of this.crew){
   if(p.alive||p.cause==='DISCONNECTED'||p.respawnAt===null||this.time+1e-7<p.respawnAt)continue;
   const targets=[...survivors].sort((a,b)=>distance(a.player.pos,p.player.pos)-distance(b.player.pos,p.player.pos));
   for(const target of targets){
    const anchor=target.player.pos,ground=floorHeight(anchor.x,anchor.z,this.time,0);
    // A jumping teammate is valid; a body already below its floor is not.
    if(anchor.y<Math.max(-8,ground+TUNING.eyeHeight-.25))continue;
    let spawn:Vec|null=null;
    for(const radius of [1.2,2.4,0]){
     for(let i=0;i<(radius?8:1);i++){
      const angle=target.input.yaw+i*Math.PI/4,x=anchor.x+Math.cos(angle)*radius,z=anchor.z+Math.sin(angle)*radius;
      const heights=[[-.25,-.25],[-.25,.25],[.25,-.25],[.25,.25]].map(([dx,dz])=>floorHeight(x+dx,z+dz,this.time,0));
      if(Math.min(...heights)<-8||Math.max(...heights)-Math.min(...heights)>.5)continue;
      spawn=v(x,Math.max(...heights)+TUNING.eyeHeight,z);break;
     }
     if(spawn)break;
    }
    if(!spawn)continue;
    p.player={pos:spawn,vel:v(),grounded:true,landedAt:-1000,hopPulse:0,jumpBuffer:0};
    p.input={...idleInput(),yaw:target.input.yaw,pitch:target.input.pitch};
    p.primary=channel();p.secondary=channel();p.wasJump=false;p.lastShotTick=-1;
    // A guest can have predicted up to two salvos per unacknowledged command
    // before dying. Reserve a gap so a new life never reuses those identities;
    // old in-flight daggers and their delayed reports keep their original keys.
    if(this.contactOwners.includes(p.id))p.shotSequence+=1024;
    p.alive=true;p.cause='';p.deathSource=undefined;p.respawnAt=null;p.life++;p.protectedUntil=this.time+CREW_RESPAWN_PROTECTION;
    break;
   }
  }
 }

 private weapon(input:Input){
  const homingAvailable=this.homingUnlocked&&this.reserve>0;
  // Homing has input priority. Entering it while streaming inherits stream mode.
  if(input.homing&&!this.secondary.wasHeld&&(this.primary.streaming||this.primary.held>16)){this.secondary.held=17;this.secondary.streaming=true;}
  this.fireChannel(this.secondary,input.homing,homingAvailable,input,true);
  const allowed=!input.homing||!homingAvailable;
  if(!this.tapFireEnabled&&this.level<5||input.fire&&!input.fireLeft&&!input.fireRight)delete this.primary.hands;
  if((this.tapFireEnabled||this.level>=5)&&(input.fireLeft||input.fireRight||this.primary.hands)){
   const hands=this.primary.hands??=[channel(),channel()];
   this.fireChannel(hands[0],!!input.fireLeft,allowed,input,false,this.level>=5?1:undefined);

  }else this.fireChannel(this.primary,input.fire,allowed,input,false);

 private fireChannel(c:FireChannel,held:boolean,allowed:boolean,input:Input,homing:boolean,hand?:number){
  if(!homing&&this.tapFireEnabled){
   // Classify each hand independently: a short release is a blast; a hold
   // becomes a beam without ever producing an opening shotgun volley.
   if(held){
    c.held++;
    if(c.held>16){
     c.streaming=true;
     if(allowed&&this.tick>=c.nextShot){this.shoot(input,false,false,hand);c.nextShot=this.tick+(this.level>=4?9:6);}
    }
   }else if(c.wasHeld){
    if(!c.streaming&&allowed&&this.tick>=c.recoverUntil){
     this.shoot(input,true,false,hand);c.lastBurst=this.tick;c.recoverUntil=this.tick+60;c.nextShot=this.tick;
    }
    c.held=0;c.streaming=false;
   }
   c.wasHeld=held;return;
  }
  if(!homing&&this.pressFireEnabled){
   // Register a blast on the press tick. Release only ends the hold; it must
   // never create a second shot or make aim depend on click duration.
   if(held){
    c.held++;
    if(!c.wasHeld&&allowed&&this.tick>=c.recoverUntil){
     this.shoot(input,true,false,hand);c.lastBurst=this.tick;
     c.recoverUntil=this.tick+60;c.nextShot=this.tick+16;
    }
    if(c.held>16){
     c.streaming=true;
     if(allowed&&this.tick>=c.nextShot){this.shoot(input,false,false,hand);c.nextShot=this.tick+(this.level>=4?9:6);}
    }
   }else{c.held=0;c.streaming=false;}
   c.wasHeld=held;return;
  }
  if(held){
   c.held++;if(c.wasHeld&&this.tick<c.recoverUntil)c.recoverUntil=Math.min(c.recoverUntil,Math.max(this.tick,c.lastBurst+45));
   if(c.held>16){c.streaming=true;if(allowed&&this.tick>=c.nextShot&&this.tick>=c.recoverUntil){this.shoot(input,false,homing,hand);c.nextShot=this.tick+(homing?12:this.level>=4?9:6);}}
  }else if(c.wasHeld){
   if(!c.streaming&&allowed&&this.tick>=c.recoverUntil){this.shoot(input,true,homing,hand);c.lastBurst=this.tick;c.recoverUntil=this.tick+60;c.nextShot=c.recoverUntil;}
   c.held=0;c.streaming=false;
  }
  c.wasHeld=held;
 }

 private updatePede(e:Enemy,dt:number,gigapedePlan?:GigapedeFlightPlan){
  if(e.kind==='gigapede'&&!this.options.legacyEnemyBehavior&&!this.options.legacyPedePursuit&&!this.options.legacyGigapedeMotion){
   updateGigapede(e,dt,this.player.pos,floorHeight(e.pos.x,e.pos.z,this.time,0),(kind,pos)=>this.emit(kind==='dust'?'spawn':kind,pos,kind==='dust'?-1:undefined,e.kind),this.gigapedeSweepsEnabled,gigapedePlan);return;
  }
  if(e.kind==='centipede'&&!this.options.legacyEnemyBehavior&&!this.options.legacyPedePursuit&&!this.options.legacyCentipedeCycle){
   (this.options.legacyCentipedeArc?updateLegacyCentipede:updateCentipede)(e,dt,{radius:this.radius,players:this.crew.length?this.crew.filter(p=>p.alive).map(p=>p.player.pos):[this.player.pos],floor:(x,z)=>floorHeight(x,z,this.time,0),random:()=>this.random(),emit:(kind,pos)=>this.emit(kind==='dust'?'spawn':kind,pos,kind==='dust'?-1:undefined,e.kind)});return;
  }
  if(this.options.legacyEnemyBehavior){this.updateLegacyPede(e,dt);return;}
  if(this.options.legacyPedePursuit){this.updateEyeSeekingPede(e,dt);return;}
  if(e.age+1e-7<e.activeAt)return;
  if(e.status==='emerging'){
   const height=e.kind==='ghostpede'?9:3,shift=height-e.pos.y;e.pos.y=height;
   for(const s of e.segments)s.pos.y+=shift;
   e.vel=v();this.emit('emerge',e.pos,undefined,e.kind,e.id);
   e.status=e.kind==='centipede'?'idle':'hunting';e.statusUntil=e.age+15;e.nextMove=e.age+3;
  }
  // Proximity burrowing follows a short hunting grace period (post-v3.1).
  const underneath=this.crew.length?this.crew.some(p=>p.alive&&p.player.pos.y<e.pos.y&&Math.hypot(p.player.pos.x-e.pos.x,p.player.pos.z-e.pos.z)<2.5):this.player.pos.y<e.pos.y&&Math.hypot(this.player.pos.x-e.pos.x,this.player.pos.z-e.pos.z)<2.5;

  if(e.status==='burrowing'){
   // Sink the whole body together; visible parts remain dangerous until buried.
   e.pos.y-=dt*8;for(const s of e.segments)s.pos.y-=dt*8;
   if(e.age+1e-7>=e.statusUntil){e.status='emerging';e.pos.y=-5;for(const s of e.segments)s.pos.y=-5;}
   return;
  }
  if(e.kind==='ghostpede'){
   const old={...e.pos};e.phase+=dt*.3;e.pos.x=Math.sin(e.phase)*(this.radius*.7);e.pos.z=Math.cos(e.phase)*(this.radius*.7);e.pos.y=9+Math.sin(e.age)*1.2;
   e.vel=v((e.pos.x-old.x)/dt,(e.pos.y-old.y)/dt,(e.pos.z-old.z)/dt);
  }else{
   if(e.status==='idle'&&e.age+1e-7>=e.nextMove)e.status='hunting';
   const goal=e.status==='idle'?v(-e.pos.z,3,e.pos.x):v(this.player.pos.x,e.pos.y,this.player.pos.z);
   const dir=direction(e.pos,goal),speed=e.kind==='gigapede'?10:8,blend=Math.min(1,dt*1.8);
   e.vel.x+=(dir.x*speed-e.vel.x)*blend;e.vel.z+=(dir.z*speed-e.vel.z)*blend;
   e.pos.x+=e.vel.x*dt;e.pos.z+=e.vel.z*dt;
   // Restore the pre-regression sweep; player jumping must not drag the head upward.
   // This waveform is replica tuning, not a measured original-game equation.
   const height=3.5+Math.sin(e.age*.7)*1.5;e.vel.y=(height-e.pos.y)/dt;e.pos.y=height;
  }
  let leader=e.pos;for(const s of e.segments){let dir=direction(leader,s.pos);if(distance(leader,s.pos)<.001)dir=v(-Math.sin(e.facingYaw),0,-Math.cos(e.facingYaw));s.pos=v(leader.x+dir.x*1.3*e.size,leader.y+dir.y*1.3*e.size,leader.z+dir.z*1.3*e.size);leader=s.pos;}
 }

  skull.vel=v(Math.sin(angle)*launchSpeed,4.5+this.random()*2.5,Math.cos(angle)*launchSpeed);

 /** Keep an outgoing stream above the crown until it has crossed its edge. */
 private towerFlightClearance(e:Enemy){
  let height=-Infinity;
  for(const tower of this.enemies){
   if(tower.hp<=0||!isSquid(tower.kind))continue;
   const scale=enemyEmergenceScale(tower),radius=towerRadius(tower.kind)*1.1*scale+e.radius+.25;
   if((e.pos.x-tower.pos.x)**2+(e.pos.z-tower.pos.z)**2<radius*radius)height=Math.max(height,tower.pos.y+(towerOutletHeight(tower.kind)+1.6)*scale+e.radius+.2);
  }
  return height;
 }
 private updateSwarm(e:Enemy,dt:number,cells:Map<string,Enemy[]>){
  if(isSkull(e.kind)&&e.statusUntil>0&&e.age<e.statusUntil){
   if(e.age<0)return;
   if(e.age<dt)this.emit('spawn',e.pos,undefined,e.kind);
   if(this.towerArchitectureEnabled&&e.towerFlight&&(!e.balance||skullTier(e.kind)!==2)){
    // Rise out of the outlet first, then turn into a stream toward the player.
    const dx=this.player.pos.x-e.pos.x,dz=this.player.pos.z-e.pos.z,length=Math.hypot(dx,dz)||1;
    const speed=(skullTier(e.kind)>=3?9:7)*clamp((e.age-.12)/.2,0,1),blend=Math.min(1,dt*10);
    e.vel.x+=(dx/length*speed-e.vel.x)*blend;e.vel.z+=(dz/length*speed-e.vel.z)*blend;
   }
   e.vel.y=Math.max(-(e.towerFlight?.descentSpeed??6)*.65,e.vel.y-(e.towerFlight?.gravity??18)*dt);
   if(this.towerArchitectureEnabled&&e.towerFlight&&e.vel.y<0&&e.pos.y<this.towerFlightClearance(e))e.vel.y=0;
   e.pos.x+=e.vel.x*dt;e.pos.y+=e.vel.y*dt;e.pos.z+=e.vel.z*dt;return;
  }
  if(isSkull(e.kind)&&e.statusUntil>0)e.status='hunting';
  const tier=isSkull(e.kind)?skullTier(e.kind):0;
  const pursuingTowerStream=this.towerArchitectureEnabled&&!!e.towerFlight&&(!e.balance||tier!==2);
  if((tier===2&&!pursuingTowerStream||e.kind==='spiderling')&&e.age>=e.nextMove){e.nextMove=e.age+(tier===2?1.6+this.random()*.9:.3+this.random()*.45);e.target=tier===2?v((this.random()-.5)*this.radius*1.5,1+this.random()*5,(this.random()-.5)*this.radius*1.5):v(this.player.pos.x+(this.random()-.5)*3,.55,this.player.pos.z+(this.random()-.5)*3);}
  const goal=tier===2&&!pursuingTowerStream||e.kind==='spiderling'?e.target:this.player.pos,d=direction(e.pos,goal),speed=tier===1?3.1:tier===2?5.6:tier===3?8:tier===4?9:Math.sin(e.age*6)>0?7:1;
  let aimX=d.x*speed,aimY=d.y*speed,aimZ=d.z*speed;
  const grounded=tier===4&&this.groundSkullsEnabled;
  if(grounded){
   // The Drake pursues across the floor, even when its target jumps. Tower
   // ejection finishes above; gravity then brings it down without a hover lane.
   const length=Math.hypot(goal.x-e.pos.x,goal.z-e.pos.z)||1;
   aimX=(goal.x-e.pos.x)/length*speed;aimZ=(goal.z-e.pos.z)/length*speed;
   aimY=e.vel.y-26*dt;
  }
  const flight=e.kind.startsWith('skull')&&!grounded?e.towerFlight:undefined;
  if(flight){
   // Individual low flight lanes blend into the actual target near contact.
   // Velocity steering brakes gently instead of snapping onto a height plane.
   const away=clamp((Math.hypot(goal.x-e.pos.x,goal.z-e.pos.z)-2)/8,0,1),lane=away*away*(3-2*away);
   const wave=Math.sin(e.age*(.9+flight.agility*.2)+e.phase);
   const floor=this.options.legacyEnemyFloor?floorHeight(e.pos.x,e.pos.z,this.time,0):enemyFloorHeight(e,this.time);
   let height=goal.y+(Math.max(floor+.9,floor+flight.height+wave*.35)-goal.y)*lane;
   if(this.towerArchitectureEnabled&&e.towerFlight)height=Math.max(height,this.towerFlightClearance(e));
   aimY=clamp((height-e.pos.y)*1.4,-flight.descentSpeed,3);
   const curl=flight.curl*wave*lane;
   aimX+=d.z*curl;aimZ-=d.x*curl;
  }
  const blend=Math.min(1,dt*(flight?flight.agility:tier===2?3:tier>=3?5:4));
  e.vel.x+=(aimX-e.vel.x)*blend;e.vel.y=grounded?aimY:e.vel.y+(aimY-e.vel.y)*(flight?Math.min(1,dt*(4+flight.agility)):blend);e.vel.z+=(aimZ-e.vel.z)*blend;
  e.pos.x+=e.vel.x*dt;e.pos.z+=e.vel.z*dt;e.pos.y=e.kind==='spiderling'?(this.options.legacyEnemyFloor?floorHeight(e.pos.x,e.pos.z,this.time,0):enemyFloorHeight(e,this.time))+.55:e.pos.y+e.vel.y*dt;
  if(e.towerFlight&&this.towerArchitectureEnabled){
   // Enlarged hovering spawners can rise while a child is already turning.
   // Resolve the outgoing lane after integration so the stream clears the maw.
   const clearance=this.towerFlightClearance(e);
   if(e.pos.y<clearance&&e.vel.y<0){e.pos.y=clearance;e.vel.y=0;}
  }
  const cx=Math.floor(e.pos.x/3),cz=Math.floor(e.pos.z/3);
  for(let x=cx-1;x<=cx+1;x++)for(let z=cz-1;z<=cz+1;z++)for(const other of cells.get(`${x},${z}`)||[]){if(other.id===e.id)continue;const dist=distance(e.pos,other.pos);if(dist<1.5&&dist>.01){const sep=direction(other.pos,e.pos);e.pos.x+=sep.x*(1.5-dist)*dt*3;e.pos.z+=sep.z*(1.5-dist)*dt*3;}}
 }

 private updateUpgradePulses(){
  for(const pulse of this.upgradePulses){
   const age=this.time-pulse.born,radius=upgradePulseRadius(age);
   for(const e of this.enemies){
    if(e.hp<=0||!this.active(e)||pulse.hit.has(e.id))continue;

    const points=e.weakpoints.length?e.weakpoints.map((s,i)=>({pos:crystalToEnemy(e,this.player.pos,s.local),hp:s.hp,index:i,weak:true,r:isSquid(e.kind)&&!legacyTowerGeometry(e)?towerCrystalRadius(e.kind,!!e.balance)*(e.balance?enemyEmergenceScale(e):1):TUNING.towerCrystalRadius})):

     [{pos:this.weakPoint(e),hp:e.hp,index:-1,weak:false,r:e.radius}];
    const point=points.find(p=>{

     const d=Math.hypot(p.pos.x-pulse.pos.x,p.pos.z-pulse.pos.z);

    });
    if(!point)continue;
    const firstEffect=this.effects.length;
    if(this.damageEnemy(e,UPGRADE_PULSE.damage[Math.min(4,pulse.level) as 2|3|4],point.pos,point.index,point.weak)>0){

     for(let i=firstEffect;i<this.effects.length;i++){this.effects[i].level=pulse.level;this.effects[i].shooterId=pulse.shooterId;}
    }
   }
   pulse.previous=radius;
  }

 }
 damageEnemy(e:Enemy,damage:number,point:Vec=e.pos,segment=-1,weak=false){

  if(segment>=0){const s=weak?e.weakpoints[segment]:e.segments[segment];applied=Math.min(s.hp,damage);s.hp-=applied;e.hp-=applied;if(applied>0&&s.hp===0)this.dropGems(point,1);}
